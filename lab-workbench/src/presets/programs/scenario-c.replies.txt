> Plan a re-trace and run it if approved
Forking the experiment for attribution only, with a larger trace buffer, then proposing the job.
```js
const cp = lab.checkpoint('before emb-churn-retrace');
const exp = lab.experiment('emb-churn').fork('emb-churn-retrace', e => e
  .purpose('attribution only: re-trace with a larger CUPTI buffer')
  .capture('trace', c => c.buffer('2GiB')));
const plan = exp.plan();
if (plan.blocked.length) { cp.restore(); return plan.blocked; }
const job = lab.start(exp.id, { reason: 'resolve the CUPTI loss in emb-churn/candidate/trace#2', modes: ['trace'] });
log(`job ${job.id} is ${job.status}: ${job.captures.length} captures on ${job.host}, ≈ ${job.budget} gpu-min — approve it in PLAN`);
await job.approved;
for await (const ev of job.events) log(ev.kind, ev.detail ?? '');
await job.done;
self.params.run = 'emb-churn-retrace/candidate/trace#2';
return lab.run('emb-churn-retrace/candidate/trace#2').coverage.trace;
```
